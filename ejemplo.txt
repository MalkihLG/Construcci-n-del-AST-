VAR salud = 100;

VAR nivel = 5;

VAR defensaTotal = defensa + (nivel * 5);
VAR poderFinal = ataqueTotal + defensaTotal;

IMPRIMIR defensaTotal;
IMPRIMIR poderFinal;

    VAR recuperacion = (salud + defensa) / 2;
    IMPRIMIR recuperacion;

    SI fuerza > velocidad {
        VAR combo = (fuerza * 3) + nivel;
        IMPRIMIR combo;
    } SINO {
        VAR escape = velocidad + defensa;
        IMPRIMIR escape;
    }

    IMPRIMIR emergencia;
}

    IMPRIMIR recompensa;
} SINO {
